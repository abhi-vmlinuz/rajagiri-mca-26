{\fontsize{14}{17}\selectfont\textbf{6.}}\ Create a function that returns a staff member's salary using the staff ID.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE FUNCTION staff_salary (p_staff_id NUMBER)
RETURN NUMBER
IS
   v_salary staff.salary%TYPE;
BEGIN
   SELECT salary INTO v_salary
   FROM staff
   WHERE staff_id = p_staff_id;

   RETURN v_salary;
END;
/

SELECT staff_salary(503) AS salary FROM dual;
\end{sqlcode}
