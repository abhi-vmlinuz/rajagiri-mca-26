{\fontsize{14}{17}\selectfont\textbf{5.}}\ Create a view showing each book category and the number of books in it.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE VIEW category_summary AS
SELECT category, COUNT(*) AS book_count
FROM book
GROUP BY category;

SELECT * FROM category_summary;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
View created.

----------------  ----------

6 rows selected.
\end{sqlout}
