{\fontsize{14}{17}\selectfont\textbf{3.}}\ Create a view named AVAILABLE\_BOOKS showing copies whose status is AVAILABLE.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE VIEW available_books AS
SELECT copy_id, book_id, branch_id, accession_no
FROM book_copy
WHERE availability_status = 'AVAILABLE';

SELECT * FROM available_books;
\end{sqlcode}
