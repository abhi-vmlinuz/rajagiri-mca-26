{\fontsize{14}{17}\selectfont\textbf{2.}}\ Create a view named ACTIVE\_MEMBERS showing members whose status is ACTIVE.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE VIEW active_members AS
SELECT member_id, full_name, member_type, city, join_date
FROM member
WHERE status = 'ACTIVE';

SELECT * FROM active_members;
\end{sqlcode}
