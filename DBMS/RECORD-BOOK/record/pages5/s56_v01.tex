{\fontsize{14}{17}\selectfont\textbf{1.}}\ Create a view named BOOK\_DETAILS showing each book's ID, title, category, publication year, and price.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE VIEW book_details AS
SELECT book_id, title, category, publication_year, price
FROM book;

SELECT * FROM book_details;
\end{sqlcode}
