{\fontsize{14}{17}\selectfont\textbf{5.}}\ Create a function that returns the total fine amount recorded for a given member ID.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE FUNCTION total_fine (p_member_id NUMBER)
RETURN NUMBER
IS
   v_total NUMBER;
BEGIN
   SELECT NVL(SUM(amount), 0) INTO v_total
   FROM fine
   WHERE member_id = p_member_id;

   RETURN v_total;
END;
/

SELECT total_fine(1012) AS total_fine FROM dual;
\end{sqlcode}
