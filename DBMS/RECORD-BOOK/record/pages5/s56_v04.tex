{\fontsize{14}{17}\selectfont\textbf{4.}}\ Create a view showing book titles and publisher names, using the relevant tables if they exist in your database.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE VIEW book_publishers AS
SELECT b.title, p.publisher_name
FROM book b
JOIN publisher p ON b.publisher_id = p.publisher_id;

SELECT * FROM book_publishers;
\end{sqlcode}
