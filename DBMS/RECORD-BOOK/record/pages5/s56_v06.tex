{\fontsize{14}{17}\selectfont\textbf{6.}}\ Create a view showing staff names and their salaries.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE OR REPLACE VIEW staff_salaries AS
SELECT full_name, salary
FROM staff;

SELECT * FROM staff_salaries;
\end{sqlcode}
