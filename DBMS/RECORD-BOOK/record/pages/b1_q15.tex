{\fontsize{14}{17}\selectfont\textbf{15.}}\ Display book ID, title, category, publication year and price for books published from 2014 onward with price above 700. Sort by price descending.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT book_id, title, category, publication_year, price
FROM book
WHERE publication_year >= 2014
  AND price > 700
ORDER BY price DESC;
\end{sqlcode}
