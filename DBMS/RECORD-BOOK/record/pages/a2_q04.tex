{\fontsize{14}{17}\selectfont\textbf{4.}}\ Insert two faculty records into the FACULTY table using a single INSERT ALL statement.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}[fontsize=\footnotesize]
INSERT ALL
  INTO faculty (faculty_id, faculty_name, designation, joining_date, salary,
                email_id, dept_id)
  VALUES ('F102', 'Ravi Kumar', 'Assistant Professor', TO_DATE('10-07-2024','DD-MM-YYYY'),
          25000, 'ravi@rajagiri.edu', 'D01')
  INTO faculty (faculty_id, faculty_name, designation, joining_date, salary,
                email_id, dept_id)
  VALUES ('F103', 'Anita Thomas', 'Professor', TO_DATE('01-06-2011','DD-MM-YYYY'),
          90000, 'anita@rajagiri.edu', 'D01')
SELECT * FROM dual;

SELECT * FROM faculty;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\scriptsize]

----------  ------------  -------------------  ------------  ------  ------------------  -------  ----------------
F102        Ravi Kumar    Assistant Professor  10-JUL-24      25000  ravi@rajagiri.edu   D01
F103        Anita Thomas  Professor            01-JUN-11      90000  anita@rajagiri.edu  D01

2 rows selected.
\end{sqlout}
