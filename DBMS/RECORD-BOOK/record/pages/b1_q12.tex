{\fontsize{14}{17}\selectfont\textbf{12.}}\ Display countries that have at least two publishers in the database. Show country and number of publishers.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT country, COUNT(*) AS no_of_publishers
FROM publisher
GROUP BY country
HAVING COUNT(publisher_id) >= 2
ORDER BY country;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
COUNTRY         NO_OF_PUBLISHERS
--------------  ----------------
United Kingdom                 2
United States                  4

2 rows selected.
\end{sqlout}
\end{minipage}\par\vspace{24pt}
