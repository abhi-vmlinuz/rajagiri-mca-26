{\fontsize{14}{17}\selectfont\textbf{6.}}\ Display the result ID, internal marks, external marks, and a calculated total obtained by adding the two marks.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT result_id, internal_marks, external_marks,
       internal_marks + external_marks AS total
FROM result;
\end{sqlcode}
