{\fontsize{14}{17}\selectfont\textbf{7.}}\ Rename the column mobile\_number in the STUDENT table to contact\_number.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
ALTER TABLE student RENAME COLUMN mobile_number TO contact_number;
DESC student;
\end{sqlcode}
