{\fontsize{14}{17}\selectfont\textbf{5.}}\ Display the reservation ID, member name, book title, and reservation status for all reservations.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT r.reservation_id, m.full_name, b.title, r.reservation_status
FROM reservation r
JOIN member m ON m.member_id = r.member_id
JOIN book b ON b.book_id = r.book_id
ORDER BY r.reservation_id;
\end{sqlcode}
