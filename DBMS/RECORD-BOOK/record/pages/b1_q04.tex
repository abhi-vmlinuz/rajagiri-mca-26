{\fontsize{14}{17}\selectfont\textbf{4.}}\ Display member name, join date and the number of complete years they have been members as of SYSDATE. Show the longest-standing members first.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT full_name, join_date,
       TRUNC(MONTHS_BETWEEN(SYSDATE, join_date) / 12) AS completed_years
FROM member
ORDER BY completed_years DESC;
\end{sqlcode}
