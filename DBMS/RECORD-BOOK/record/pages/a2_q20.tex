{\fontsize{14}{17}\selectfont\textbf{20.}}\ Delete all events for which no registration fee is charged.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
DELETE FROM event WHERE registration_fee IS NULL OR registration_fee = 0;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
1 row deleted.
Commit complete.
\end{sqlout}
