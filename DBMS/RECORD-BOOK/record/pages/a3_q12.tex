{\fontsize{14}{17}\selectfont\textbf{12.}}\ Display students whose CGPA is between 7 and 9.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT student_id, student_name, cgpa, dept_id
FROM student WHERE cgpa BETWEEN 7 AND 9;
\end{sqlcode}
