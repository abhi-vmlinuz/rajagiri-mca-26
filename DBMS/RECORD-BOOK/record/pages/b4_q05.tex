{\fontsize{14}{17}\selectfont\textbf{5.}}\ Perform the following operations in a single transaction: update the status of a member, create a savepoint named member\_update, update the price of a book, roll back to member\_update, and commit the transaction.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
UPDATE member SET status = 'SUSPENDED' WHERE member_id = 1007;
SAVEPOINT member_update;
UPDATE book SET price = 1200 WHERE book_id = 4007;
ROLLBACK TO member_update;
COMMIT;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
1 row updated.
Savepoint created.
1 row updated.
Rollback complete.
Commit complete.
\end{sqlout}
\end{minipage}\par\vspace{24pt}
