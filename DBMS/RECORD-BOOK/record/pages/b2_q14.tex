{\fontsize{14}{17}\selectfont\textbf{14.}}\ Display publisher name, book title, and price using a RIGHT OUTER JOIN. Include all books, even if a matching publisher record is not found.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT p.publisher_name, b.title, b.price
FROM publisher p
RIGHT JOIN book b ON b.publisher_id = p.publisher_id
ORDER BY b.book_id;
\end{sqlcode}
