{\fontsize{14}{17}\selectfont\textbf{5.}}\ Create a view named LOAN\_DETAILS to display member name, book title, issue date, due date, and loan status by joining the appropriate tables.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE VIEW loan_details AS
SELECT m.full_name, b.title, l.issue_date, l.due_date, l.loan_status
FROM loan l
JOIN member m ON m.member_id = l.member_id
JOIN book_copy c ON c.copy_id = l.copy_id
JOIN book b ON b.book_id = c.book_id;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
View created.
\end{sqlout}
