{\fontsize{14}{17}\selectfont\textbf{13.}}\ Display book title, copy ID, and availability status using a RIGHT OUTER JOIN. Include all physical copies in BOOK\_COPY.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT b.title, c.copy_id, c.availability_status
FROM book b
RIGHT JOIN book_copy c ON c.book_id = b.book_id
ORDER BY c.copy_id;
\end{sqlcode}
