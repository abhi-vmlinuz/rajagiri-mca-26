{\fontsize{14}{17}\selectfont\textbf{18.}}\ Display book title, copy ID, and branch name using a NATURAL JOIN between BOOK\_COPY and LIBRARY\_BRANCH.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT b.title, c.copy_id, l.branch_name
FROM book_copy c
NATURAL JOIN library_branch l
JOIN book b ON b.book_id = c.book_id
ORDER BY c.copy_id;
\end{sqlcode}
