\textbf{Table Name:} ENROLLMENT \hspace{1em plus 1fill}
\textbf{Description:} Records the courses enrolled by each student.

\vspace{8pt}

\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\textwidth}{|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|}
\hline
\textbf{Column Name} & \textbf{Description} & \textbf{Data Type} & \textbf{Constraint} \\
\hline
enrollment\_id & Enrollment ID & CHAR(5) & PRIMARY KEY \\
\hline
student\_id & Student ID & CHAR(5) & FOREIGN KEY REFERENCES student (student\_id) \\
\hline
course\_id & Course Code & CHAR(5) & FOREIGN KEY REFERENCES course (course\_id) \\
\hline
enrollment\_date & Enrollment Date & DATE & DEFAULT SYSDATE \\
\hline
academic\_year & Academic Year & VARCHAR2(9) & -- \\
\hline
\end{tabularx}

\vspace{14pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE TABLE enrollment (
    enrollment_id   CHAR(5)      PRIMARY KEY,
    student_id      CHAR(5)      REFERENCES student(student_id),
    course_id       CHAR(5)      REFERENCES course(course_id),
    enrollment_date DATE         DEFAULT SYSDATE,
    academic_year   VARCHAR2(9),
    UNIQUE (student_id, course_id)
);
\end{sqlcode}
