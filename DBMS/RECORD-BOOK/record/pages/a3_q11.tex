{\fontsize{14}{17}\selectfont\textbf{11.}}\ Display events for which the registration fee is greater than 500.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT * FROM event WHERE registration_fee > 500;
\end{sqlcode}
