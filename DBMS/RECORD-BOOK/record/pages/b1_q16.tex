{\fontsize{14}{17}\selectfont\textbf{16.}}\ Display books whose title contains either the word 'Data' or the word 'Learning', ignoring case. Show title, category, year, pages and rating.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT title, category, publication_year, pages, rating
FROM book
WHERE LOWER(title) LIKE '%data%'
   OR LOWER(title) LIKE '%learning%'
ORDER BY book_id;
\end{sqlcode}

\vspace{18pt}
