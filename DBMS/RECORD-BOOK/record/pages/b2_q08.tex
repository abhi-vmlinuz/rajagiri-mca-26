{\fontsize{14}{17}\selectfont\textbf{8.}}\ Display all books and their physical copy IDs. Include books that do not have any physical copies.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT b.title, c.copy_id
FROM book b
LEFT JOIN book_copy c ON c.book_id = b.book_id
ORDER BY b.book_id, c.copy_id;
\end{sqlcode}
