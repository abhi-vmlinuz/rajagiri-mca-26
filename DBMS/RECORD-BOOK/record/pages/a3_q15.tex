{\fontsize{14}{17}\selectfont\textbf{15.}}\ Display faculty members whose email addresses belong to the rajagiri.edu domain.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT faculty_id, faculty_name, email_id
FROM faculty WHERE email_id LIKE '%@rajagiri.edu';
\end{sqlcode}
