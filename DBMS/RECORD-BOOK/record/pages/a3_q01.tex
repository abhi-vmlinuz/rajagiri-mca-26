{\fontsize{14}{17}\selectfont\textbf{1.}}\ Display the different faculty designations without duplicates.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT DISTINCT designation FROM faculty;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
DESIGNATION        
-------------------
Assistant Professor
Professor
Associate Professor

3 rows selected.
\end{sqlout}
