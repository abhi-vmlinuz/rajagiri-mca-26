{\fontsize{14}{17}\selectfont\textbf{3.}}\ Display the staff name, role, and branch name for all staff members.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT s.full_name, s.role_name, br.branch_name
FROM staff s
JOIN library_branch br ON br.branch_id = s.branch_id
ORDER BY s.staff_id;
\end{sqlcode}
