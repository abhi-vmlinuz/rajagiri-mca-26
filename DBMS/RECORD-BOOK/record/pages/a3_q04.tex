{\fontsize{14}{17}\selectfont\textbf{4.}}\ Display the faculty name, monthly salary, and annual salary. Calculate annual salary as monthly salary multiplied by 12.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT faculty_name, salary, salary * 12 AS annual_salary
FROM faculty;
\end{sqlcode}
