{\fontsize{14}{17}\selectfont\textbf{11.}}\ Display each publisher's name, country and age in years in 2026 (2026 - established\_year). Show the oldest publishers first.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT publisher_name, country,
       (EXTRACT(YEAR FROM SYSDATE) - established_year) AS age
FROM publisher
ORDER BY age DESC;
\end{sqlcode}
