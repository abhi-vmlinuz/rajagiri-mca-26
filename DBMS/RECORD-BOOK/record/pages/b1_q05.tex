{\fontsize{14}{17}\selectfont\textbf{5.}}\ Display the names and email addresses of members whose name contains the letter 'a' anywhere, ignoring case, and whose account is ACTIVE.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT LOWER(full_name) AS name, email
FROM member
WHERE LOWER(full_name) LIKE '%a%'
  AND status = 'ACTIVE';
\end{sqlcode}
