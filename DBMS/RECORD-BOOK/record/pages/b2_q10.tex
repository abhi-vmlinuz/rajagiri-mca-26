{\fontsize{14}{17}\selectfont\textbf{10.}}\ Display all authors and the titles of books associated with them through BOOK\_AUTHOR. Include authors who are not linked to any book.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT a.author_name, b.title
FROM author a
LEFT JOIN book_author ba ON ba.author_id = a.author_id
LEFT JOIN book b ON b.book_id = ba.book_id
ORDER BY a.author_id, b.title;
\end{sqlcode}
