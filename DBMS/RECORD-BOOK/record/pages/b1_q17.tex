{\fontsize{14}{17}\selectfont\textbf{17.}}\ Display books with more than 500 pages and a rating of at least 4.5. Add a computed column named PRICE\_PER\_100\_PAGES showing the approximate price per 100 pages, rounded to two decimals.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT title, pages, rating,
       ROUND((price / pages) * 100, 2) AS price_per_100_pages
FROM book
WHERE pages > 500
  AND rating >= 4.5
ORDER BY pages DESC;
\end{sqlcode}
