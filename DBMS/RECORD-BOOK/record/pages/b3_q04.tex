{\fontsize{14}{17}\selectfont\textbf{4.}}\ Create a view named BOOK\_PUBLISHER\_DETAILS to display book title, publisher name, and price by joining BOOK and PUBLISHER.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE VIEW book_publisher_details AS
SELECT b.title, p.publisher_name, b.price
FROM book b
JOIN publisher p ON p.publisher_id = b.publisher_id;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
View created.
\end{sqlout}
