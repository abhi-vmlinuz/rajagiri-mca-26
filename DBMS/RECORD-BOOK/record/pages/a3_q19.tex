{\fontsize{14}{17}\selectfont\textbf{19.}}\ Display student names along with the number of characters in each name.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT student_name, LENGTH(student_name) AS name_length
FROM student;
\end{sqlcode}
