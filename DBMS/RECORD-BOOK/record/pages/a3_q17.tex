{\fontsize{14}{17}\selectfont\textbf{17.}}\ Display all student names in uppercase.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT UPPER(student_name) AS student_name FROM student;
\end{sqlcode}
