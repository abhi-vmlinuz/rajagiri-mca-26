{\fontsize{14}{17}\selectfont\textbf{13.}}\ Display courses offered in Semesters 1, 3, or 5 using the IN operator.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT course_id, course_title, semester
FROM course WHERE semester IN (1, 3, 5);
\end{sqlcode}

\vspace{18pt}

8 rows selected.
\end{sqlout}
