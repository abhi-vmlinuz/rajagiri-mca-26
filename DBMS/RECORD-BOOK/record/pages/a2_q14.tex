{\fontsize{14}{17}\selectfont\textbf{14.}}\ Change the office location of the Computer Science department to Golden Aureole Block.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}[fontsize=\footnotesize]
UPDATE department SET office_location = 'Golden Aureole Block'
WHERE dept_name = 'Computer Science';

SELECT dept_id, dept_name, office_location FROM department
WHERE dept_name = 'Computer Science';
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
1 row updated.

DEPT_ID  DEPT_NAME         OFFICE_LOCATION     
-------  ----------------  --------------------
D01      Computer Science  Golden Aureole Block

1 row selected.
\end{sqlout}
\end{minipage}\par\vspace{24pt}
