{\fontsize{14}{17}\selectfont\textbf{5.}}\ Display the faculty name, current salary, and salary after a 10\% increase without modifying the table.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT faculty_name, salary, salary * 1.10 AS increased_salary
FROM faculty;
\end{sqlcode}
