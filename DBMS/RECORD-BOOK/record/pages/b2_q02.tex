{\fontsize{14}{17}\selectfont\textbf{2.}}\ Display each library member's name, member type, and the name of the branch to which the member belongs.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT m.full_name, m.member_type, br.branch_name
FROM member m
JOIN library_branch br ON br.branch_id = m.branch_id
ORDER BY m.member_id;
\end{sqlcode}
