{\fontsize{14}{17}\selectfont\textbf{2.}}\ Add a column named address with datatype VARCHAR2(200) to the STUDENT table.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
ALTER TABLE student ADD (address VARCHAR2(200));
DESC student;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}
 Name                                      Null?    Type
 ----------------------------------------- -------- ----------------------------
 STUDENT_ID                                NOT NULL CHAR(5)
 STUDENT_NAME                              NOT NULL VARCHAR2(50)
 DOB                                                DATE
 GENDER                                             CHAR(1)
 MOBILE_NUMBER                                      VARCHAR2(10)
 EMAIL_ID                                           VARCHAR2(50)
 DEPT_ID                                            VARCHAR2(10)
 ADMISSION_DATE                                     DATE
 CGPA                                               NUMBER(4,2)
 ADDRESS                                            VARCHAR2(200)
\end{sqlout}
