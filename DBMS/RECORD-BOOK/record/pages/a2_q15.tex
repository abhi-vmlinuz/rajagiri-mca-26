{\fontsize{14}{17}\selectfont\textbf{15.}}\ Update the credits of the course Database Management Systems from 4 to 5.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}[fontsize=\footnotesize]
UPDATE course SET credits = 5
WHERE course_title = 'Database Management Systems';

SELECT course_id, course_title, credits FROM course
WHERE course_title = 'Database Management Systems';
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
1 row updated.

COURSE_ID  COURSE_TITLE                 CREDITS
---------  ---------------------------  -------
C101       Database Management Systems        5

1 row selected.
\end{sqlout}
