{\fontsize{14}{17}\selectfont\textbf{7.}}\ Insert a course without mentioning the column names. Ensure that the values are entered in the correct table-column order.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
INSERT INTO course
VALUES ('C102', 'Operating Systems', 3, 8, 'D01', 'F102');

SELECT * FROM course;
\end{sqlcode}
