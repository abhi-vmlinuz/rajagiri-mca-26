{\fontsize{14}{17}\selectfont\textbf{16.}}\ Set the result status to PASS for results with total marks greater than or equal to 50, and FAIL for the remaining records.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
UPDATE result
SET result_status = CASE WHEN total_marks >= 50 THEN 'PASS' ELSE 'FAIL' END;

SELECT result_id, total_marks, result_status FROM result;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
1 row updated.

RESULT_ID  TOTAL_MARKS  RESULT_STATUS
---------  -----------  -------------
R101                40  FAIL

1 row selected.
\end{sqlout}
\end{minipage}\par\vspace{24pt}
