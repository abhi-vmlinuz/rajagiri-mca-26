{\fontsize{14}{17}\selectfont\textbf{3.}}\ Display all members who are either STUDENT or RESEARCHER and whose maximum borrowing limit is at least 4 books. Sort by max\_books descending.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT member_id, full_name, member_type, max_books
FROM member
WHERE member_type IN ('RESEARCHER', 'STUDENT')
  AND max_books >= 4
ORDER BY max_books DESC;
\end{sqlcode}
