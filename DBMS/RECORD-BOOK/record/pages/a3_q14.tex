{\fontsize{14}{17}\selectfont\textbf{14.}}\ Display students whose names begin with the letter A.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT student_id, student_name, cgpa
FROM student WHERE student_name LIKE 'A%';
\end{sqlcode}
