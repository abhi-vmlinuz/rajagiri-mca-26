{\fontsize{14}{17}\selectfont\textbf{2.}}\ Display the different semesters in which courses are offered.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT DISTINCT semester FROM course;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
SEMESTER
--------
       3
       1
       5
       7

4 rows selected.
\end{sqlout}
