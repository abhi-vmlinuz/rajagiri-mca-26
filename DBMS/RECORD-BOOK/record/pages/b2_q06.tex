{\fontsize{14}{17}\selectfont\textbf{6.}}\ Display the fine ID, member name, loan ID, fine amount, and payment status for fines recorded in the library.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT f.fine_id, m.full_name, f.loan_id, f.amount, f.payment_status
FROM fine f
JOIN member m ON m.member_id = f.member_id
ORDER BY f.fine_id;
\end{sqlcode}
