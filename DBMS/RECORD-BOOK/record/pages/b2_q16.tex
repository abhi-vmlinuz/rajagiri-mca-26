{\fontsize{14}{17}\selectfont\textbf{16.}}\ Display all books and all reservations, including books that have never been reserved and reservation records without a matching book.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT b.title, r.reservation_id
FROM book b
FULL OUTER JOIN reservation r ON r.book_id = b.book_id
ORDER BY b.book_id, r.reservation_id;
\end{sqlcode}
