{\fontsize{14}{17}\selectfont\textbf{2.}}\ List the members whose outstanding amount is between 50 and 300, inclusive. Show member ID, name, outstanding amount and status in descending order of outstanding amount.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT member_id, full_name, outstanding_amt, status
FROM member
WHERE outstanding_amt BETWEEN 50 AND 300
ORDER BY outstanding_amt DESC;
\end{sqlcode}

\vspace{18pt}
