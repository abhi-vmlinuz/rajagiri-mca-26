{\fontsize{14}{17}\selectfont\textbf{17.}}\ Display all members and all fine records, including members without fines and fine records without a matching member.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT m.full_name, f.fine_id
FROM member m
FULL OUTER JOIN fine f ON f.member_id = m.member_id
ORDER BY m.member_id, f.fine_id;
\end{sqlcode}
