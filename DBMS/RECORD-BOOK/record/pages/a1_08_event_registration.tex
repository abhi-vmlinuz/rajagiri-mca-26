\textbf{Table Name:} EVENT\_REGISTRATION \hspace{1em plus 1fill}
\textbf{Description:} Records student registrations for each event.

\vspace{8pt}

\renewcommand{\arraystretch}{1.4}
\begin{tabularx}{\textwidth}{|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|>{\raggedright\arraybackslash}X|}
\hline
\textbf{Column Name} & \textbf{Description} & \textbf{Data Type} & \textbf{Constraint} \\
\hline
registration\_id & Registration ID & CHAR(5) & PRIMARY KEY \\
\hline
event\_id & Event ID & CHAR(5) & FOREIGN KEY REFERENCES event (event\_id) \\
\hline
student\_id & Student ID & CHAR(5) & FOREIGN KEY REFERENCES student (student\_id) \\
\hline
registration\_date & Registration Date & DATE & DEFAULT SYSDATE \\
\hline
participation\_status & Participation Status & VARCHAR2(15) & CHECK (REGISTERED, ATTENDED, ABSENT) \\
\hline
\end{tabularx}

\vspace{14pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE TABLE event_registration (
    registration_id      CHAR(5)       PRIMARY KEY,
    event_id             CHAR(5)       REFERENCES event(event_id),
    student_id           CHAR(5)       REFERENCES student(student_id),
    registration_date    DATE          DEFAULT SYSDATE,
    participation_status VARCHAR2(15)
        CHECK (participation_status IN ('REGISTERED', 'ATTENDED', 'ABSENT'))
);
\end{sqlcode}
