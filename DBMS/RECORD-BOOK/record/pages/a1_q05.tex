{\fontsize{14}{17}\selectfont\textbf{5.}}\ Modify the size of the student\_name column to VARCHAR2(100).\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
ALTER TABLE student MODIFY (student_name VARCHAR2(100));
DESC student;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}
