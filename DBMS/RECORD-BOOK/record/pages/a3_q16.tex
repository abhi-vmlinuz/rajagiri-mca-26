{\fontsize{14}{17}\selectfont\textbf{16.}}\ Display students whose CGPA has not been entered.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT student_id, student_name
FROM student WHERE cgpa IS NULL;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
STUDENT_ID  STUDENT_NAME
----------  ------------
S102        Neha Thomas

1 row selected.
\end{sqlout}
\end{minipage}\par\vspace{24pt}
