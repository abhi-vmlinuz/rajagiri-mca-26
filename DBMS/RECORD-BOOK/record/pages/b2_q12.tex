{\fontsize{14}{17}\selectfont\textbf{12.}}\ Display branch name, city, and staff name using a RIGHT OUTER JOIN. Include all branches, even if a branch has no staff.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT br.branch_name, br.city, s.full_name
FROM staff s
RIGHT JOIN library_branch br ON br.branch_id = s.branch_id
ORDER BY br.branch_id, s.staff_id;
\end{sqlcode}
