{\fontsize{14}{17}\selectfont\textbf{19.}}\ Display member name, book title, copy accession number, issue date, and loan status for books that have been loaned.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT m.full_name, b.title, c.accession_no, l.issue_date, l.loan_status
FROM loan l
JOIN member m ON m.member_id = l.member_id
JOIN book_copy c ON c.copy_id = l.copy_id
JOIN book b ON b.book_id = c.book_id
ORDER BY l.loan_id;
\end{sqlcode}
