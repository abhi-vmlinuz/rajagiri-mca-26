{\fontsize{14}{17}\selectfont\textbf{2.}}\ Create a view named MEMBER\_DETAILS to display the member ID, member name, member type, and status of all library members.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE VIEW member_details AS
SELECT member_id, full_name, member_type, status
FROM member;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
View created.
\end{sqlout}
