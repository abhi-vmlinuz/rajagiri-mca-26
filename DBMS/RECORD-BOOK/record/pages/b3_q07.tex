{\fontsize{14}{17}\selectfont\textbf{7.}}\ Display books from the BOOK\_PUBLISHER\_DETAILS view whose price is greater than 500.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT title, publisher_name, price
FROM book_publisher_details
WHERE price > 500
ORDER BY price DESC;
\end{sqlcode}
