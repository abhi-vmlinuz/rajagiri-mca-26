{\fontsize{14}{17}\selectfont\textbf{11.}}\ Display all publishers and the titles of books published by them. Include publishers that have not published any books.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT p.publisher_name, b.title
FROM publisher p
LEFT JOIN book b ON b.publisher_id = p.publisher_id
ORDER BY p.publisher_id, b.book_id;
\end{sqlcode}
