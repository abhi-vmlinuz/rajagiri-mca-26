{\fontsize{14}{17}\selectfont\textbf{3.}}\ Create a view named MEMBER\_BRANCH\_DETAILS to display member name, member type, branch name, and city by joining MEMBER and LIBRARY\_BRANCH.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
CREATE VIEW member_branch_details AS
SELECT m.full_name, m.member_type, br.branch_name, br.city
FROM member m
JOIN library_branch br ON br.branch_id = m.branch_id;
\end{sqlcode}

\vspace{18pt}

\codeheading{OUTPUT:}

\begin{sqlout}[fontsize=\small]
View created.
\end{sqlout}
