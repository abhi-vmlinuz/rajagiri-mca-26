{\fontsize{14}{17}\selectfont\textbf{7.}}\ Display all library branches and the names of members registered at each branch. Include branches that have no members.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT br.branch_name, m.full_name
FROM library_branch br
LEFT JOIN member m ON m.branch_id = br.branch_id
ORDER BY br.branch_id, m.member_id;
\end{sqlcode}
