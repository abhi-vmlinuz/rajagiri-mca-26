{\fontsize{14}{17}\selectfont\textbf{9.}}\ Display all members and their loan IDs. Include members who have never borrowed a book.\par
\endgroup\vspace{12pt}

\codeheading{CODE:}

\begin{sqlcode}
SELECT m.full_name, l.loan_id
FROM member m
LEFT JOIN loan l ON l.member_id = m.member_id
ORDER BY m.member_id, l.loan_id;
\end{sqlcode}
