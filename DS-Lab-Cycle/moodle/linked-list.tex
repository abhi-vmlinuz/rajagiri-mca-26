\section*{Problem Statement}

\noindent
\fcolorbox{codeframe}{codebg}{%
\begin{minipage}{0.97\textwidth}
\vspace{0.4em}
\textbf{Question:}\\[0.3em]
Write a program to create and manage a singly linked list containing student records. Each record should include the student's name, roll number, and marks. Implement the following operations:
\begin{enumerate}
    \item \textbf{Insert} a new student record into the linked list.
    \item \textbf{Delete} a student record using the roll number and name.
    \item \textbf{Search} for and display a particular student's details.
    \item \textbf{Display} all student records.
\end{enumerate}
\vspace{0.2em}
\end{minipage}%
}
